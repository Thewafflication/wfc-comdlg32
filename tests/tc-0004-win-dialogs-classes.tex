\def\TCID{TC-0004}
\def\TCTitle{A program uses the dialog classes with scripted answers}
\def\TCPurpose{%
  Verify in a running WFC program that scripted answers reach the dialog
  classes (\texttt{ColorDialog}, \texttt{OpenFileDialog},
  \texttt{SaveFileDialog}, \texttt{FontDialog}) and that a scripted cancel
  leaves a property unchanged.}
\def\TCPriority{\TCPriorityMedium}

\def\TCRequirementRef{REQ-0003, REQ-0004, REQ-0005}
\def\TCDesignRef{Not applicable.}
\def\TCFeatureRef{%
  Corpus case \texttt{tests/corpus/207-win-dialogs-classes} (project,
  form, UI script and expected output); runner
  \texttt{tests/run-tc-0004-win-dialogs-classes.ps1}. WFC registers the same
  case as \texttt{TC-MP0002-corpus-207-win-dialogs-classes}.}
\def\TCTestTechnique{%
  Scenario test through the public interface of the program; the output is
  compared exactly.}

\def\TCPreconditions{%
  A WFC executable (\texttt{wfc.exe}) is available; the runner finds it on
  \texttt{PATH}, in the environment variable \texttt{WFC\_EXE}, or in the
  parameter \texttt{-Wfc}.}
\def\TCEnvironment{%
  Windows or any host WFC runs on; the interpreter runs the project, with
  no window shown and no system dialog opened (the UI script answers).}
\def\TCAssumptions{%
  The WFC executable reads the UI script named in the project
  (\texttt{WfcUiScript}) and supports the dialog classes.}

\def\TCInputData{%
  The UI script: the answers colour 65280, open
  \texttt{D:$\backslash$games$\backslash$save.smbk}, save
  \texttt{E:$\backslash$out$\backslash$game.smbk}, font
  \texttt{Courier,12,1,0} and colour \texttt{cancel}, then a click on the
  button of the form.}
\def\TCInitialState{%
  The form sets colour 255, the filter \texttt{Text|*.txt|All|*.*} and the
  font name Arial before it shows the dialogs; a final colour dialog starts
  with colour 7.}

\def\TCProcedure{%
  \begin{enumerate}
    \item Run \texttt{wfc --interpret
      tests/corpus/207-win-dialogs-classes.vbp}.
    \item Capture the standard output and the exit code.
    \item Compare the output, after the line ends are normalised and the
      ends are stripped, with
      \texttt{tests/corpus/207-win-dialogs-classes.expected}.
  \end{enumerate}}
\def\TCExpectedResult{%
  WFC exits with code 0 and the output equals the expected file: the colour
  dialog returns OK with colour 65280; the open dialog returns OK with the
  scripted path, filter index 1 and the safe file name \texttt{save.smbk};
  the save dialog returns OK with the scripted path; the font dialog returns
  OK with Courier, 12 and bold; the colour constants print as expected; the
  last colour dialog is cancelled and the colour stays 7.}
\def\TCPostConditions{Not applicable; the test changes nothing.}

\TCTemplate
